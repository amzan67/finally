function updateMenu(modelFile, name, description, price) {
    const viewer = document.getElementById('main-viewer');
    
    // 1. Kill the current rendering to prevent the freeze
    viewer.src = ""; 

    // 2. Update the text details immediately
    document.getElementById('item-name').innerText = name;
    document.getElementById('item-description').innerText = description;
    document.getElementById('item-price').innerText = price;

    // 3. Wait 100ms for the browser to clear memory, then load the new model
    setTimeout(() => {
        viewer.setAttribute('src', modelFile);
        
        // 4. Force the viewer to update its internal state
        viewer.dismissPoster(); 
        console.log("Loading: " + modelFile);
    }, 100);
}
